    <script>
        let cart = [];

        // Filter products by category
        function filterProducts(category, button) {
            document.querySelectorAll(".filters button")
                .forEach(btn => btn.classList.remove("active"));

            button.classList.add("active");

            document.querySelectorAll(".product[data-category]")
                .forEach(product => {
                    product.style.display =
                        category === "All" ||
                        product.dataset.category === category
                        ? "" : "none";
                });

            document.getElementById("search").value = "";
        }

        // Search products
        function searchProducts() {
            const search = document.getElementById("search")
                .value.toLowerCase();

            document.querySelectorAll(".product[data-category]")
                .forEach(product => {
                    product.style.display =
                        product.innerText.toLowerCase().includes(search)
                        ? "" : "none";
                });

            document.querySelectorAll(".filters button")
                .forEach(btn => btn.classList.remove("active"));
        }

        // Add product to cart
        function addToCart(name, price) {
            cart.push({ name: name, price: price });
            updateCart();
        }

        // Update cart display
        function updateCart() {
            document.getElementById("cart-count").textContent = cart.length;

            const cartItems = document.getElementById("cart-items");
            const totalElement = document.getElementById("cart-total");

            if (cart.length === 0) {
                cartItems.textContent = "Your cart is empty.";
                totalElement.textContent = "Total: ₹0";
                return;
            }

            cartItems.innerHTML = "";

            let total = 0;

            cart.forEach((item, index) => {
                total += item.price;

                const row = document.createElement("p");
                row.textContent = item.name + " - ₹" +
                    item.price.toLocaleString("en-IN") +
                    " ";

                const removeButton = document.createElement("button");
                removeButton.textContent = "Remove";
                removeButton.onclick = function() {
                    cart.splice(index, 1);
                    updateCart();
                };

                row.appendChild(removeButton);
                cartItems.appendChild(row);
            });

            totalElement.textContent =
                "Total: ₹" + total.toLocaleString("en-IN");
        }

        // Open cart section
        function showCart() {
            document.getElementById("cart-section")
                .scrollIntoView({ behavior: "smooth" });
        }

        // Clear cart
        function clearCart() {
            cart = [];
            updateCart();
        }

        // Show product details
        function showDetails(name, category, description, price) {
            document.getElementById("detail-name").textContent = name;
            document.getElementById("detail-category").textContent =
                "Category: " + category;
            document.getElementById("detail-description").textContent =
                description;
            document.getElementById("detail-price").textContent = price;

            document.getElementById("details")
                .scrollIntoView({ behavior: "smooth" });
        }
    </script>

    <script>
        // Room prices per night
        const prices = {
            "Standard Room": 2000,
            "Deluxe Room": 3500,
            "Luxury Suite": 5500
        };

        const checkin = document.getElementById("checkin");
        const checkout = document.getElementById("checkout");
        const roomType = document.getElementById("roomType");

        // Set today's date as the earliest check-in
        const today = new Date();
        const todayString =
            today.getFullYear() + "-" +
            String(today.getMonth() + 1).padStart(2, "0") + "-" +
            String(today.getDate()).padStart(2, "0");

        checkin.min = todayString;
        checkout.min = todayString;

        // Select a room from its card
        function selectRoom(room) {
            roomType.value = room;
            document.getElementById("booking").scrollIntoView({
                behavior: "smooth"
            });
            calculatePrice();
        }

        // Update checkout's earliest allowed date
        checkin.addEventListener("change", function() {
            checkout.min = checkin.value;

            if (checkout.value && checkout.value <= checkin.value) {
                checkout.value = "";
            }

            calculatePrice();
        });

        checkout.addEventListener("change", calculatePrice);
        roomType.addEventListener("change", calculatePrice);

        // Calculate estimated total
        function calculatePrice() {
            const room = roomType.value;
            const start = new Date(checkin.value + "T00:00:00");
            const end = new Date(checkout.value + "T00:00:00");

            if (room && checkin.value && checkout.value && end > start) {
                const nights = Math.round(
                    (end - start) / (1000 * 60 * 60 * 24)
                );

                const total = nights * prices[room];

                document.getElementById("totalPrice").textContent =
                    "Estimated Total: ₹" + total.toLocaleString("en-IN") +
                    " for " + nights + " night(s)";
            } else {
                document.getElementById("totalPrice").textContent = "";
            }
        }

        // Demonstration reservation form
        document.getElementById("bookingForm").addEventListener(
            "submit", function(event) {
                event.preventDefault();

                if (checkout.value <= checkin.value) {
                    document.getElementById("message").textContent =
                        "Please select a checkout date after check-in.";
                    return;
                }

                const name = document.getElementById("name").value;

                document.getElementById("message").textContent =
                    "Thank you, " + name +
                    "! Your reservation request has been recorded in this demo.";

                this.reset();
                checkout.min = todayString;
                document.getElementById("totalPrice").textContent = "";
            }
        );
    </script>
